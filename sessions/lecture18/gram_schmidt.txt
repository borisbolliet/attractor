Lecture 18, Gram-Schmidt as a matrix identity Y = QR: each column of Y minus its components along the earlier columns of Q, then normalised.
    python -m doctest -v sessions/lecture18/gram_schmidt.txt

>>> import numpy as np
>>> def gram_schmidt(Y):
...     n, l = Y.shape
...     Q, R = np.zeros((n, l)), np.zeros((l, l))
...     for j in range(l):
...         R[:j, j] = Q[:, :j].T @ Y[:, j]    # along q_0, ..., q_(j-1)
...         y = Y[:, j] - Q[:, :j] @ R[:j, j]  # orthogonal to them
...         R[j, j] = np.linalg.norm(y)
...         Q[:, j] = y / R[j, j]
...     return Q, R
>>> Y = np.array([[3.0, 1.0], [4.0, 2.0], [0.0, 2.0]])
>>> Q, R = gram_schmidt(Y)
>>> R.round(4)
array([[5.    , 2.2   ],
       [0.    , 2.0396]])
>>> np.allclose(Q @ R, Y), np.allclose(Q.T @ Q, np.eye(2))
(True, True)
>>> Q2, R2 = np.linalg.qr(Y)                   # Householder reflections
>>> np.allclose(np.abs(R2), R), np.allclose(np.abs(Q2.T @ Q), np.eye(2))
(True, True)
