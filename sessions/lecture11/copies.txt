>>> import numpy as np
>>> from attractor.trajectory import Trajectory
>>> tr = Trajectory(np.arange(6.0), np.arange(12.0).reshape(6, 2))
>>> rv, ev = tr.time_reversed(), tr.every(2)
>>> rv.y.base is tr.y, ev.y.base is tr.y, rv.t.base is None
(True, True, True)
>>> rv.y[0, 0] = 99.0
>>> tr.y[-1]
array([99., 11.])
>>> w = tr.window(1.0, 3.0)
>>> w, np.shares_memory(w.y, tr.y), w.y.base is None
(Trajectory(3 points, t in [1, 3]), False, True)
